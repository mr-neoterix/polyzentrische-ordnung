% Pandoc-Vorlage für das Manuskript „Die polyzentrische Ordnung“.
%
% Gesetzt wird mit LuaLaTeX. Die Vorlage ersetzt die mitgelieferte Vorlage
% von Pandoc absichtlich vollständig: Der Text ist reine Prosa ohne Formeln,
% Tabellen, Bilder und Listen, deshalb kommt der Satz mit wenigen Paketen
% aus – das hält die Installation klein und den Lauf schnell.

  fontsize=11pt,

  parskip=false,

% --------------------------------------------------------------------------
% Schrift
% --------------------------------------------------------------------------
% Pagella (ein Palatino-Nachbau) ist für deutsche Fließtextprosa gebaut:
% offene Formen, kräftiges ß, ruhige Umlaute.

% Angesprochen werden die Schriftdateien, nicht die Schriftnamen: So findet
% LuaTeX sie über den TeX-Baum auch dort, wo kein Fontconfig-Zwischenspeicher
% gepflegt ist – etwa in einem frisch aufgesetzten Bauläufer.

\setmainfont{texgyrepagella}[
  Extension = .otf,
  UprightFont = *-regular,
  ItalicFont = *-italic,
  BoldFont = *-bold,
  BoldItalicFont = *-bolditalic,
  Ligatures = TeX,
]
\setsansfont{texgyreheros}[
  Extension = .otf,
  UprightFont = *-regular,
  ItalicFont = *-italic,
  BoldFont = *-bold,
  BoldItalicFont = *-bolditalic,
  Ligatures = TeX,
  Scale = MatchLowercase,
]
\setmonofont{lmmono10}[
  Extension = .otf,
  UprightFont = *-regular,
  ItalicFont = *-italic,
  Scale = MatchLowercase,
]

% --------------------------------------------------------------------------
% Gliederung
% --------------------------------------------------------------------------
% Nichts wird nummeriert: Die Kapitel tragen ihre Zählung ausgeschrieben im
% Namen („Neuntes Kapitel“), die Teile ebenso. Das Inhaltsverzeichnis führt
% nur Teile und Kapitel; die Abschnitte stehen als Lesezeichen im PDF.

\setcounter{secnumdepth}{-\maxdimen}
\setcounter{tocdepth}{0}

% KOMA setzt Überschriften und Verzeichniseinträge serifenlos. In einem Buch,
% dessen Text eine Antiqua ist, wirkt das wie ein Fremdkörper.
\setkomafont{disposition}{\normalfont\bfseries}
\setkomafont{section}{\normalfont\large\bfseries}
\setkomafont{chapterentry}{\normalfont}
\setkomafont{chapterentrypagenumber}{\normalfont}

\makeatletter

% Ein Kapitelanfang: Bezeichnung als Zeile darüber, Titel darunter.
\newcommand{\kapitel}[2]{%
  \clearpage
  \phantomsection
  \thispagestyle{plain}%
  \vspace*{0.08\textheight}%
  \begingroup
    \raggedright
    {\normalfont\large\scshape\color{akzent}#1\par}%
    \vspace{0.7em}%
    {\normalfont\Huge\bfseries#2\par}%
  \endgroup
  \vspace{2.4em}%
  \addcontentsline{toc}{chapter}{#1 – #2}%
  \markboth{#2}{#2}%
  \@afterindentfalse\@afterheading
}

% Eine Teilseite: allein auf der Seite, ohne Kolumnentitel.
\newcommand{\teil}[1]{%
  \clearpage
  \phantomsection
  \thispagestyle{empty}%
  \null\vspace*{0.24\textheight}%
  \begingroup
    \centering
    {\normalfont\LARGE\bfseries\color{akzent}#1\par}%
  \endgroup
  \addcontentsline{toc}{part}{#1}%
  \markboth{#1}{#1}%
  \clearpage
}

% Ein Vorspannabschnitt: wie ein Kapitel, aber schlichter.
\newcommand{\vorspann}[1]{%
  \clearpage
  \phantomsection
  \thispagestyle{plain}%
  \vspace*{0.04\textheight}%
  \begingroup
    \raggedright
    {\normalfont\LARGE\bfseries#1\par}%
  \endgroup
  \vspace{1.6em}%
  \addcontentsline{toc}{chapter}{#1}%
  \markboth{#1}{#1}%
  \@afterindentfalse\@afterheading
}

\newcommand{\hauptteil}{\cleardoublepage\mainmatter}

% Der Belegapparat am Kapitelende: kleiner gesetzt, mit Abstand statt
% Einzug, damit er als Apparat und nicht als Fließtext gelesen wird.
\newcounter{belegblock}
\newcommand{\belegeanfang}{%
  \par\bigskip
  \stepcounter{belegblock}%
  \pdfbookmark[1]{Belege}{belege.\thebelegblock}%
  \noindent\textcolor{leise}{\rule{0.3\linewidth}{0.4pt}}\par
  \nobreak\vspace{0.9em}%
  \nobreak{\normalfont\scshape\color{akzent}Belege}\par
  \nobreak\vspace{0.5em}%
  \begingroup
    \small
    \setlength{\parindent}{0pt}%
    \setlength{\parskip}{0.55em}%
}
\newcommand{\belegeende}{\par\endgroup}

\makeatother
